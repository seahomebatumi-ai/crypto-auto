START_MS = FREEZE_MS - 30*86400*1000

SYMS = ["BTCUSDT","SUIUSDT","ONDOUSDT","LINKUSDT","RENDERUSDT","NEARUSDT","YFIUSDT",

BASE = "https://fapi.binance.com"
UA = {"User-Agent":"analyst-hunt/1.0"}

def fetch(path, retries=3):
    url = BASE + path
    last_exc = None
    for i in range(retries):
        req = urllib.request.Request(url, headers=UA)
        try:
            with urllib.request.urlopen(req, timeout=20) as resp:
                return resp.status, resp.geturl(), resp.read()
        except urllib.error.HTTPError as e:
            return e.code, url, e.read()
        except Exception as e:
            last_exc = e
            time.sleep(1+i)
    return None, url, str(last_exc).encode()

def num(x):
    try:
        v = float(x)
        if math.isfinite(v):
            return v
    except Exception:
        pass
    return None

results = {}
for sym in SYMS:
    rec = {"symbol": sym, "reads": {}}
    st, host, body = fetch(f"/futures/data/openInterestHist?symbol={sym}&period=4h&limit=180")
    rec["reads"]["oi"] = {"status": st, "host": host, "bytes": len(body)}
    oi_rows = []
    if st == 200:
        arr = json.loads(body)
        if isinstance(arr, list) and arr and "sumOpenInterest" in arr[0] and "timestamp" in arr[0]:
            for r in arr:
                t, v = r.get("timestamp"), num(r.get("sumOpenInterest"))
                if t is not None and v is not None and v > 0:
                    oi_rows.append((int(t), v))
        else:
            rec["reads"]["oi"]["shape_error"] = True
    oi_rows.sort(); oi_rows = [r for r in oi_rows if r[0] <= FREEZE_MS]

    st2, host2, body2 = fetch(f"/futures/data/topLongShortPositionRatio?symbol={sym}&period=4h&limit=180")
    rec["reads"]["large"] = {"status": st2, "host": host2, "bytes": len(body2)}
    large_rows = []
    if st2 == 200:
        arr = json.loads(body2)
        if isinstance(arr, list) and arr and "longShortRatio" in arr[0] and "timestamp" in arr[0]:
            for r in arr:
                t, v = r.get("timestamp"), num(r.get("longShortRatio"))
                if t is not None and v is not None and v > 0:
                    large_rows.append((int(t), v))
        else:
            rec["reads"]["large"]["shape_error"] = True
    large_rows.sort(); large_rows = [r for r in large_rows if r[0] <= FREEZE_MS]

    st3, host3, body3 = fetch(f"/futures/data/globalLongShortAccountRatio?symbol={sym}&period=4h&limit=180")
    rec["reads"]["crowd"] = {"status": st3, "host": host3, "bytes": len(body3)}
    crowd_rows = []
    if st3 == 200:
        arr = json.loads(body3)
        if isinstance(arr, list) and arr and "longShortRatio" in arr[0] and "timestamp" in arr[0]:
            for r in arr:
                t, v = r.get("timestamp"), num(r.get("longShortRatio"))

                    crowd_rows.append((int(t), v))
        else:
            rec["reads"]["crowd"]["shape_error"] = True
    crowd_rows.sort(); crowd_rows = [r for r in crowd_rows if r[0] <= FREEZE_MS]

    st4, host4, body4 = fetch(f"/fapi/v1/fundingRate?symbol={sym}&startTime={START_MS}&limit=1000")
    rec["reads"]["funding"] = {"status": st4, "host": host4, "bytes": len(body4)}
    funding_rows = []
    if st4 == 200:
        arr = json.loads(body4)
        if isinstance(arr, list) and arr and "fundingTime" in arr[0] and "fundingRate" in arr[0]:
            for r in arr:
                t = r.get("fundingTime")
                try:

                    if not math.isfinite(v): v = None
                except Exception:

                if t is not None and v is not None:

        elif arr:

    funding_rows.sort(); funding_rows = [r for r in funding_rows if r[0] <= FREEZE_MS]

    rec["counts"] = {"oi": len(oi_rows), "large": len(large_rows), "crowd": len(crowd_rows), "funding": len(funding_rows)}

    unread_reason = None
    if rec["reads"]["oi"].get("shape_error") or rec["reads"]["large"].get("shape_error") or rec["reads"]["crowd"].get("shape_error"):

            idx = max(1, min(m, math.ceil(p*m)))

            crowd_class = "long"
        elif crowd_now <= qCrowd25 and funding_now <= qF25:
            crowd_class = "short"

    rec["calc"] = {"dL_now": dL_now, "dC_now": dC_now, "dQ_now": dQ_now,
        "qL25": qL25, "qL50": qL50, "qL75": qL75, "qC25": qC25, "qC50": qC50, "qC75": qC75,
