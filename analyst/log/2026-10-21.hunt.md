import json, math, time, urllib.request, urllib.error, datetime

FREEZE = datetime.datetime(2026,10,21,12,23,7,tzinfo=datetime.timezone.utc)
FREEZE_MS = int(FREEZE.timestamp()*1000)
START_MS = FREEZE_MS - 30*24*3600*1000

def fetch(url):
    req = urllib.request.Request(url, headers={'User-Agent':'curl/8.0'})
    try:
        with urllib.request.urlopen(req, timeout=20) as r:
            return r.status, r.geturl(), r.read()
    except urllib.error.HTTPError as e:
        return e.code, url, e.read()
    except Exception as e:
        return None, str(e), b''

def get_json(url):
    status, host, body = fetch(url)
    try:
        data = json.loads(body)
    except Exception:
        data = None
    return status, host, data, len(body)

results = {}
log = []
base = "https://fapi.binance.com"
for sym in SYMS:
    urls = {
        "oi": f"{base}/futures/data/openInterestHist?symbol={sym}&period=4h&limit=180",
        "large": f"{base}/futures/data/topLongShortPositionRatio?symbol={sym}&period=4h&limit=180",
        "crowd": f"{base}/futures/data/globalLongShortAccountRatio?symbol={sym}&period=4h&limit=180",
        "funding": f"{base}/fapi/v1/fundingRate?symbol={sym}&startTime={START_MS}&limit=1000",
    }
    raw = {}
    for key, url in urls.items():
        status, host, data, nbytes = get_json(url)
        raw[key] = data
        first_keys = sorted(data[0].keys()) if isinstance(data, list) and data and isinstance(data[0], dict) else None
        log.append({"sym": sym, "endpoint": key, "status": status, "host": host, "bytes": nbytes,
                     "n": len(data) if isinstance(data, list) else None, "first_keys": first_keys})
        time.sleep(0.08)
    results[sym] = raw

json.dump({"results": results, "log": log}, open("/tmp/positioning_raw.json","w"))
