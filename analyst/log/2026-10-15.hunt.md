#!/usr/bin/env python3
"""Positioning read (ANALYST-INSTRUCTIONS.md sec.16) - run once, text archived verbatim in hunt record."""
import urllib.request, json, sys, time, math, statistics

FREEZE_MS = 1792075219000  # 2026-10-15T14:40:19Z

COINS = {
 "BTC":"BTCUSDT","SUI":"SUIUSDT","ONDO":"ONDOUSDT","LINK":"LINKUSDT","RENDER":"RENDERUSDT",
 "NEAR":"NEARUSDT","YFI":"YFIUSDT","AAVE":"AAVEUSDT","AVAX":"AVAXUSDT","FET":"FETUSDT",
 "ENA":"ENAUSDT","TAO":"TAOUSDT","GRAM":"GRAMUSDT","XRP":"XRPUSDT","ADA":"ADAUSDT",
 "TRX":"TRXUSDT","SOL":"SOLUSDT","BCH":"BCHUSDT","HYPE":"HYPEUSDT","SKY":"SKYUSDT",
 "ETH":"ETHUSDT","HBAR":"HBARUSDT","XLM":"XLMUSDT","ALGO":"ALGOUSDT","BNB":"BNBUSDT",
 "ZEC":"ZECUSDT","UNI":"UNIUSDT","XMR":"XMRUSDT","LIT":"LITUSDT","MORPHO":"MORPHOUSDT","ARB":"ARBUSDT",
 "WAL":"WALUSDT","STRK":"STRKUSDT","AVA":"AVAUSDT","RSR":"RSRUSDT","1000RATS":"1000RATSUSDT",
 "CAP":"CAPUSDT","LAB":"LABUSDT","BEAT":"BEATUSDT",
}

K = 42  # 7 days of 4h rows

def fetch(url):
    req = urllib.request.Request(url, headers={"User-Agent":"curl/8.0"})
    t0=time.time()
    try:
        with urllib.request.urlopen(req, timeout=20) as resp:
            body = resp.read()
            return resp.status, resp.geturl(), body, len(body)
    except urllib.error.HTTPError as e:
        return e.code, url, e.read() if e.fp else b"", 0
    except Exception as e:
        return None, url, str(e).encode(), 0

def to_f(v):
    try:
        f = float(v)
        if not math.isfinite(f): return None
        return f
    except Exception:
        return None

def get_series(sym, kind):
    url_map = {
        "oi": f"https://fapi.binance.com/futures/data/openInterestHist?symbol={sym}&period=4h&limit=180",
        "large": f"https://fapi.binance.com/futures/data/topLongShortPositionRatio?symbol={sym}&period=4h&limit=180",
        "crowd": f"https://fapi.binance.com/futures/data/globalLongShortAccountRatio?symbol={sym}&period=4h&limit=180",
    }
    status, landed, body, n = fetch(url_map[kind])
    rec = {"status": status, "host": landed, "bytes": n}
    if status != 200:
        rec["rows"] = 0
        return [], rec
    try:
        doc = json.loads(body)
    except Exception:
        rec["rows"] = -1
        return [], rec
    if not isinstance(doc, list):
        rec["rows"] = -1
        return [], rec
    out = []
    field = "sumOpenInterest" if kind=="oi" else "longShortRatio"
    for row in doc:
        if not isinstance(row, dict) or "timestamp" not in row or field not in row:
            continue
        ts = row.get("timestamp")
        v = to_f(row.get(field))
        if v is None or v <= 0:
            continue
        out.append((ts, v))
    out.sort(key=lambda t: t[0])
    out = [r for r in out if r[0] <= FREEZE_MS]
    rec["rows"] = len(out)
    rec["first_keys"] = sorted(doc[0].keys()) if doc and isinstance(doc[0], dict) else []
    return out, rec

def get_funding(sym):
    start = FREEZE_MS - 30*24*3600*1000
    url = f"https://fapi.binance.com/fapi/v1/fundingRate?symbol={sym}&startTime={start}&limit=1000"
    status, landed, body, n = fetch(url)
    rec = {"status": status, "host": landed, "bytes": n}
    if status != 200:
        rec["rows"]=0
        return [], rec
    try:
        doc = json.loads(body)
    except Exception:
        rec["rows"]=-1
        return [], rec
    out=[]
    for row in doc:
        if not isinstance(row,dict) or "fundingTime" not in row or "fundingRate" not in row:
            continue
        ts=row.get("fundingTime")
        v=to_f(row.get("fundingRate"))
        if v is None:
            continue
        out.append((ts,v))
    out.sort(key=lambda t:t[0])
    out=[r for r in out if r[0]<=FREEZE_MS]
    rec["rows"]=len(out)
    rec["first_keys"]=sorted(doc[0].keys()) if doc and isinstance(doc[0],dict) else []
    return out, rec

def quartiles(vals):
    if not vals: return (None,None,None)
    s=sorted(vals)
    m=len(s)
    def rank(p):
        idx = math.ceil(p*m)
        idx = max(1,min(m,idx))
        return s[idx-1]
    return rank(0.25), rank(0.50), rank(0.75)

def ln_change(series, k):
    # series: list of (ts, v) sorted oldest first
    out=[]
    for i in range(k, len(series)):
        v0 = series[i-k][1]
        v1 = series[i][1]
        if v0>0 and v1>0:
            out.append((series[i][0], math.log(v1/v0)))
    return out

results = {}
log = []

for sym, pair in COINS.items():
    entry = {"sym": sym, "pair": pair}
    oi_series, oi_rec = get_series(pair, "oi")
    large_series, large_rec = get_series(pair, "large")
    crowd_series, crowd_rec = get_series(pair, "crowd")
    funding_series, funding_rec = get_funding(pair)
    entry["reads"] = {"oi": oi_rec, "large": large_rec, "crowd": crowd_rec, "funding": funding_rec}

    # 14 days = 84 rows of 4h min; funding >=14 days back
    unread = False
    reasons=[]
    if len(oi_series) < 84: unread=True; reasons.append(f"oi rows={len(oi_series)}")
    if len(large_series) < 84: unread=True; reasons.append(f"large rows={len(large_series)}")
    if len(crowd_series) < 84: unread=True; reasons.append(f"crowd rows={len(crowd_series)}")
    if funding_series:
        span_days = (funding_series[-1][0]-funding_series[0][0])/86400000.0
        if span_days < 14: unread=True; reasons.append(f"funding span={span_days:.1f}d")
    else:
        unread=True; reasons.append("funding rows=0")

    entry["unread"] = unread
    entry["unread_reasons"] = reasons
    if unread:
        results[sym]=entry
        continue

    dL = ln_change(large_series, K)
    dC = ln_change(crowd_series, K)
    dQ = ln_change(oi_series, K)

    if not dL or not dC or not dQ:
        entry["unread"]=True
        entry["unread_reasons"]=["insufficient change series"]
        results[sym]=entry
        continue

    dL_now = dL[-1][1]; dC_now = dC[-1][1]; dQ_now = dQ[-1][1]
    q25L,q50L,q75L = quartiles([v for _,v in dL])
    q25C,q50C,q75C = quartiles([v for _,v in dC])
    crowd_now = crowd_series[-1][1]
    q25crowd,q50crowd,q75crowd = quartiles([v for _,v in crowd_series])
    funding_now = funding_series[-1][1]
    q25f,q50f,q75f = quartiles([v for _,v in funding_series])

    entry["dL_now"]=dL_now; entry["dC_now"]=dC_now; entry["dQ_now"]=dQ_now
    entry["q_dL"]=(q25L,q50L,q75L); entry["q_dC"]=(q25C,q50C,q75C)
    entry["crowd_now"]=crowd_now; entry["q_crowd"]=(q25crowd,q50crowd,q75crowd)
    entry["funding_now"]=funding_now; entry["q_funding"]=(q25f,q50f,q75f)

    large_class = None
    if dL_now >= q75L and dC_now <= q50C and dQ_now > 0:
        large_class = "long"
    elif dL_now <= q25L and dC_now >= q50C and dQ_now > 0:
        large_class = "short"

    crowd_class = None
    if crowd_now >= q75crowd and funding_now >= q75f:
        crowd_class = "long"
    elif crowd_now <= q25crowd and funding_now <= q25f:
        crowd_class = "short"

    entry["large_class"]=large_class
    entry["crowd_class"]=crowd_class
    results[sym]=entry

json.dump(results, open("/tmp/positioning_results.json","w"), default=str, indent=1)
print("DONE", len(results))
